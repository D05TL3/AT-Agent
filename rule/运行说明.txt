ARG NODE_IMAGE=node:22-bookworm-slim
ARG BASE_IMAGE=ubuntu:24.04

FROM ${NODE_IMAGE} AS node-runtime

FROM ${BASE_IMAGE}

ARG OPENCODE_VERSION=1.18.29
USER root

LABEL org.opencontainers.image.title="AWD OpenCode Security Runtime" \
      org.opencontainers.image.description="Restricted OpenCode runtime with security analysis tooling" \
      org.opencontainers.image.version="${OPENCODE_VERSION}" \
      awd.runtime.contract="1"

ENV DEBIAN_FRONTEND=noninteractive \
    HOME=/tmp/home \
    XDG_CONFIG_HOME=/tmp/home/.config \
    XDG_DATA_HOME=/tmp/home/.local/share \
    XDG_CACHE_HOME=/tmp/home/.cache \
    PATH=/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin

# 官方 Node.js 镜像将 node、npm 和 npx 安装在 /usr/local。
COPY --from=node-runtime /usr/local/ /usr/local/

# 基础工具：curl、Git、jq、ripgrep、Python、压缩工具。
# 安全工具：
# - 网络探测：Nmap、Netcat、dig、OpenSSH Client
# - Web 测试：Gobuster、SQLMap
# - 二进制分析：Binutils、GDB、Checksec、Strace、xxd
# - 静态分析：ExifTool、YARA、SQLite
#
# 仅安装明确列出的软件包且不安装推荐依赖；构建结束后删除 APT 索引。
# 发行版检查可防止 BASE_IMAGE 被误改为非 Ubuntu 镜像。
RUN grep -q '^ID=ubuntu$' /etc/os-release \
    && apt-get update \
    && apt-get install -y --no-install-recommends \
       bash \
       binutils \
       ca-certificates \
       checksec \
       coreutils \
       curl \
       dnsutils \
       file \
       gdb \
       git \
       gobuster \
       jq \
       libimage-exiftool-perl \
       libstdc++6 \
       netcat-openbsd \
       nmap \
       openssh-client \
       procps \
       python3 \
       python3-requests \
       ripgrep \
       sed \
       shellcheck \
       sqlmap \
       sqlite3 \
       strace \
       tini \
       unzip \
       xxd \
       yara \
       zip \
    && apt-get clean \
    && rm -rf /var/lib/apt/lists/*

# OpenCode 使用固定版本，禁止在比赛运行期间自动升级。
RUN npm install --global --omit=dev "opencode-ai@${OPENCODE_VERSION}" \
    && npm cache clean --force \
    && test "$(opencode --version)" = "${OPENCODE_VERSION}"

ENV OPENCODE_CONFIG=/opt/awd/opencode.json \
    OPENCODE_CONFIG_DIR=/tmp/home/.config/opencode \
    OPENCODE_CONFIG_CONTENT="{\"autoupdate\":false,\"share\":\"disabled\",\"server\":{\"mdns\":false},\"permission\":\"allow\"}"

COPY config/opencode.json /opt/awd/opencode.json
COPY scripts/entrypoint.sh /usr/local/bin/awd-opencode-entrypoint
COPY scripts/runtime-self-check.sh /usr/local/bin/runtime-self-check

RUN chmod 0755 /usr/local/bin/awd-opencode-entrypoint /usr/local/bin/runtime-self-check \
    && mkdir -p /workspace/agent /workspace/input /workspace/output \
    && chmod 0755 /workspace /workspace/agent /workspace/input /workspace/output \
    && runtime-self-check --build
 
WORKDIR /workspace/agent
EXPOSE 4096

USER 65534:65534

ENTRYPOINT ["/usr/bin/tini", "--", "/usr/local/bin/awd-opencode-entrypoint"]
CMD ["opencode", "serve", "--hostname", "0.0.0.0", "--port", "4096"]
